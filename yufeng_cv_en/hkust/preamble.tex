% Custom commands
\newcommand{\resumeItem}[2]{
  \item\small{
    \textbf{#1}{: #2 \vspace{-2pt}}
  }
}

\newcommand{\resumeSubheading}[4]{
  \vspace{-1pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{p{0.68\textwidth}@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small #3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-5pt}
}

\newcommand{\resumeItemPlain}[1]{
  \item\small{
    {#1 \vspace{-2pt}}
  }
}

\newcommand{\resumeSubItem}[2]{\resumeItem{#1}{#2}\vspace{-4pt}}

% One-line skill row: label + content
\newcommand{\skillrow}[2]{

}

% Publication entry: title (left, bold) + venue (right), authors below, role below that
\newcommand{\pubentry}[4]{%
  \begin{tabular*}{\textwidth}[t]{@{}p{0.70\textwidth}@{\extracolsep{\fill}}r@{}}
    \small\textbf{#1} & \small\textit{#2}
  \end{tabular*}\par\vspace{-2pt}
  {\small\textit{#3}}\par\vspace{-1pt}
  {\small #4}\par\vspace{4pt}
}
